#include <reg52.h>
#include <intrins.h>

// =========================================================
// 1. PIN DEFINITIONS (Matched to A2 Board Schematic)
// =========================================================
sbit RS = P2^6;   // LCD_RS on P2.6
sbit RW = P2^5;   // LCD_RW on P2.5
sbit EN = P2^7;   // LCD_EN on P2.7

#define LCD_DATA_PORT P0

// =========================================================
// 2. FUNCTION PROTOTYPES
// =========================================================
void delay_ms(unsigned int ms);
void delay_us(unsigned int us);
void lcd_cmd_nibble(unsigned char nibble);
void lcd_cmd(unsigned char cmd);
void lcd_data(unsigned char ch);
void lcd_string(char *str);
void lcd_init(void);

// =========================================================
// 3. MAIN FUNCTION
// =========================================================
void main(void) {
    // Initialize 16x2 LCD in 4-bit mode
    lcd_init();

    // Line 1 output
    lcd_string("LCD Initialized!");

    // Move to Line 2 (Address 0xC0)
    lcd_cmd(0xC0);
    lcd_string("System Ready...");

    while (1) {
        // Main loop
    }
}

// =========================================================
// 4. TIMING DELAYS
// =========================================================
void delay_ms(unsigned int ms) {
    unsigned int i, j;
    for (i = 0; i < ms; i++) {
        for (j = 0; j < 300; j++); // Adjusted loop timing for STC core
    }
}

void delay_us(unsigned int us) {
    while (us--) {
        _nop_();
        _nop_();
    }
}

// =========================================================
// 5. LCD LOW-LEVEL DRIVERS
// =========================================================
void lcd_cmd_nibble(unsigned char nibble) {
    RS = 0;                  // Command mode
    RW = 0;                  // Write mode
    
    // Mask lower 4 bits of P0 and write nibble to P0.4-P0.7
    LCD_DATA_PORT = (LCD_DATA_PORT & 0x0F) | ((nibble & 0x0F) << 4);
    
    EN = 1;
    delay_us(10);
    EN = 0;
    delay_us(10);
}

void lcd_cmd(unsigned char cmd) {
    lcd_cmd_nibble(cmd >> 4);   // Send High Nibble
    lcd_cmd_nibble(cmd);        // Send Low Nibble
    delay_ms(3);                // Wait for execution
}

void lcd_data(unsigned char ch) {
    RS = 1;                  // Data mode
    RW = 0;                  // Write mode
    
    // Send High Nibble
    LCD_DATA_PORT = (LCD_DATA_PORT & 0x0F) | (ch & 0xF0);
    EN = 1; delay_us(10); EN = 0;
    
    // Send Low Nibble
    LCD_DATA_PORT = (LCD_DATA_PORT & 0x0F) | ((ch & 0x0F) << 4);
    EN = 1; delay_us(10); EN = 0;
    
    delay_us(100);
}

void lcd_string(char *str) {
    while (*str) {
        lcd_data(*str++);
    }
}

// =========================================================
// 6. HD44780 4-BIT INITIALIZATION
// =========================================================
void lcd_init(void) {
    delay_ms(50); // Power-on stabilization delay

    // Soft reset sequence to sync LCD state machine
    lcd_cmd_nibble(0x03);
    delay_ms(10);
    lcd_cmd_nibble(0x03);
    delay_ms(1);
    lcd_cmd_nibble(0x03);
    delay_ms(1);

    // Force switch to 4-bit mode
    lcd_cmd_nibble(0x02); 
    delay_ms(2);

    // Configuration commands
    lcd_cmd(0x28); // 4-bit mode, 2 lines, 5x8 dots
    lcd_cmd(0x0C); // Display ON, Cursor OFF
    lcd_cmd(0x01); // Clear Screen
    delay_ms(5);
    lcd_cmd(0x06); // Entry mode: Increment cursor
}
