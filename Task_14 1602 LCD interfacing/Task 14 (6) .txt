#include <reg52.h>
#include <intrins.h>

// =========================================================
// 1. HARDWARE PIN MAPPINGS (A2 Board Schematic)
// =========================================================
sbit RS = P2^6;   // LCD Register Select (P2.6)
sbit RW = P2^5;   // LCD Read/Write (P2.5)
sbit EN = P2^7;   // LCD Enable (P2.7)

#define LCD_DATA_PORT P0  // Data pins connected to P0.4 - P0.7 in 4-bit mode

// =========================================================
// 2. CUSTOM CHARACTER BITMAPS (HEX FORMAT FOR KEIL C51)
// =========================================================

// Char 0: Heart
unsigned char code heart[8] = { 0x00, 0x0A, 0x1F, 0x1F, 0x0E, 0x04, 0x00, 0x00 };

// Char 1: Bell
unsigned char code bell[8] = { 0x04, 0x0E, 0x0E, 0x0E, 0x1F, 0x00, 0x04, 0x00 };

// Char 2: Space Invader / Alien
unsigned char code invader[8] = { 0x0E, 0x1F, 0x15, 0x1F, 0x0A, 0x15, 0x00, 0x00 };

// Char 3: Tick / Checkmark
unsigned char code tick[8] = { 0x00, 0x01, 0x03, 0x16, 0x1C, 0x08, 0x00, 0x00 };

// Char 4: Speaker / Volume
unsigned char code speaker[8] = { 0x01, 0x03, 0x0F, 0x0F, 0x0F, 0x03, 0x01, 0x00 };

// Char 5: Music Note
unsigned char code music[8] = { 0x02, 0x03, 0x02, 0x0E, 0x1E, 0x0C, 0x00, 0x00 };

// Char 6: Skull / Lightbulb
unsigned char code skull[8] = { 0x00, 0x0E, 0x15, 0x1F, 0x0E, 0x0A, 0x00, 0x00 };

// Char 7: Lock
unsigned char code lock[8] = { 0x0E, 0x11, 0x11, 0x1F, 0x1B, 0x1B, 0x1F, 0x00 };

// =========================================================
// 3. FUNCTION DECLARATIONS
// =========================================================
void delay_ms(unsigned int ms);
void delay_us(unsigned int us);
void lcd_cmd_nibble(unsigned char nibble);
void lcd_cmd(unsigned char cmd);
void lcd_data(unsigned char ch);
void lcd_string(char *str);
void lcd_init(void);
void lcd_create_char(unsigned char location, unsigned char *charmap);

// =========================================================
// 4. MAIN APPLICATION
// =========================================================
void main(void) {
    unsigned char i;

    // Initialize LCD in 4-Bit Mode
    lcd_init();

    // Load Custom Glyphs into CGRAM Locations 0 to 7
    lcd_create_char(0, heart);
    lcd_create_char(1, bell);
    lcd_create_char(2, invader);
    lcd_create_char(3, tick);
    lcd_create_char(4, speaker);
    lcd_create_char(5, music);
    lcd_create_char(6, skull);
    lcd_create_char(7, lock);

    // Print Line 1: Title
    lcd_cmd(0x80);              // Move to Line 1, Position 0
    lcd_string("Custom Character");

    // Print Line 2: Display all 8 custom characters spaced out
    lcd_cmd(0xC0);              // Move to Line 2, Position 0
    for (i = 0; i < 8; i++) {
        lcd_data(i);            // Print character from CGRAM address i
        lcd_data(' ');          // Space between characters
    }

    while (1) {
        // Main loop
    }
}

// =========================================================
// 5. DRIVER & CGRAM HELPER FUNCTIONS
// =========================================================

// Store custom character array to LCD CGRAM
void lcd_create_char(unsigned char location, unsigned char *charmap) {
    unsigned char i;
    location &= 0x07;                // CGRAM only supports locations 0..7
    lcd_cmd(0x40 | (location << 3)); // Set CGRAM Address pointer
    for (i = 0; i < 8; i++) {
        lcd_data(charmap[i]);
    }
}

void delay_ms(unsigned int ms) {
    unsigned int i, j;
    for (i = 0; i < ms; i++) {
        for (j = 0; j < 300; j++);
    }
}

void delay_us(unsigned int us) {
    while (us--) {
        _nop_();
        _nop_();
    }
}

void lcd_cmd_nibble(unsigned char nibble) {
    RS = 0;
    RW = 0;
    LCD_DATA_PORT = (LCD_DATA_PORT & 0x0F) | ((nibble & 0x0F) << 4);
    EN = 1; delay_us(10); EN = 0; delay_us(10);
}

void lcd_cmd(unsigned char cmd) {
    lcd_cmd_nibble(cmd >> 4);
    lcd_cmd_nibble(cmd);
    delay_ms(3);
}

void lcd_data(unsigned char ch) {
    RS = 1;
    RW = 0;
    LCD_DATA_PORT = (LCD_DATA_PORT & 0x0F) | (ch & 0xF0);
    EN = 1; delay_us(10); EN = 0;
    
    LCD_DATA_PORT = (LCD_DATA_PORT & 0x0F) | ((ch & 0x0F) << 4);
    EN = 1; delay_us(10); EN = 0;
    
    delay_us(100);
}

void lcd_string(char *str) {
    while (*str) {
        lcd_data(*str++);
    }
}

void lcd_init(void) {
    delay_ms(50);
    
    // 4-Bit Mode Initialization Protocol
    lcd_cmd_nibble(0x03); delay_ms(10);
    lcd_cmd_nibble(0x03); delay_ms(1);
    lcd_cmd_nibble(0x03); delay_ms(1);
    lcd_cmd_nibble(0x02); delay_ms(2);

    lcd_cmd(0x28); // 4-bit mode, 2-line display, 5x8 font
    lcd_cmd(0x0C); // Display ON, Cursor OFF
    lcd_cmd(0x01); // Clear Display RAM
    delay_ms(5);
    lcd_cmd(0x06); // Automatic cursor movement
}
