#include <reg52.h>
#include <intrins.h>

// =========================================================
// 1. HARDWARE PIN MAPPINGS (A2 Board Schematic)
// =========================================================
sbit RS = P2^6;   // LCD Register Select (P2.6)
sbit RW = P2^5;   // LCD Read/Write (P2.5)
sbit EN = P2^7;   // LCD Enable (P2.7)

// 4-bit data bus uses P0.4 - P0.7 connected to DB4 - DB7
#define LCD_DATA_PORT P0 

// =========================================================
// 2. FUNCTION DECLARATIONS
// =========================================================
void delay_ms(unsigned int ms);
void delay_us(unsigned int us);
void lcd_cmd_nibble(unsigned char nibble);
void lcd_cmd(unsigned char cmd);
void lcd_data(unsigned char ch);
void lcd_string(char *str);
void lcd_init(void);

// =========================================================
// 3. MAIN APPLICATION
// =========================================================
void main(void) {
    // Step 1: Initialize LCD1602 in 4-bit mode
    lcd_init();

    // Step 2: Display string on First Line
    lcd_cmd(0x80);              // Force cursor to Line 1, Position 0
    lcd_string("4-Bit LCD Mode");

    // Step 3: Display string on Second Line
    lcd_cmd(0xC0);              // Force cursor to Line 2, Position 0
    lcd_string("Task 5 Complete!");

    while (1) {
        // Main loop execution completed
    }
}

// =========================================================
// 4. TIMING UTILITIES
// =========================================================
void delay_ms(unsigned int ms) {
    unsigned int i, j;
    for (i = 0; i < ms; i++) {
        for (j = 0; j < 300; j++); // Calibrated loop count for STC89 core
    }
}

void delay_us(unsigned int us) {
    while (us--) {
        _nop_();
        _nop_();
    }
}

// =========================================================
// 5. LOW-LEVEL 4-BIT LCD DRIVERS
// =========================================================
void lcd_cmd_nibble(unsigned char nibble) {
    RS = 0; // Command mode
    RW = 0; // Write mode
    
    // Output 4 bits onto P0.4 - P0.7 while leaving P0.0 - P0.3 intact
    LCD_DATA_PORT = (LCD_DATA_PORT & 0x0F) | ((nibble & 0x0F) << 4);
    
    // Enable Pulse (Latch Data)
    EN = 1;
    delay_us(10);
    EN = 0;
    delay_us(10);
}

void lcd_cmd(unsigned char cmd) {
    lcd_cmd_nibble(cmd >> 4);   // Send High Nibble
    lcd_cmd_nibble(cmd);        // Send Low Nibble
    delay_ms(3);                // Command processing delay
}

void lcd_data(unsigned char ch) {
    RS = 1; // Data mode
    RW = 0; // Write mode
    
    // Send High Nibble
    LCD_DATA_PORT = (LCD_DATA_PORT & 0x0F) | (ch & 0xF0);
    EN = 1; delay_us(10); EN = 0;
    
    // Send Low Nibble
    LCD_DATA_PORT = (LCD_DATA_PORT & 0x0F) | ((ch & 0x0F) << 4);
    EN = 1; delay_us(10); EN = 0;
    
    delay_us(100);
}

void lcd_string(char *str) {
    while (*str) {
        lcd_data(*str++);
    }
}

// =========================================================
// 6. HD44780 4-BIT INITIALIZATION SEQUENCE
// =========================================================
void lcd_init(void) {
    delay_ms(50); // Power-on startup time

    // Forced Software Reset Procedure
    lcd_cmd_nibble(0x03);
    delay_ms(10);
    lcd_cmd_nibble(0x03);
    delay_ms(1);
    lcd_cmd_nibble(0x03);
    delay_ms(1);

    // Switch Interface to 4-bit Mode
    lcd_cmd_nibble(0x02); 
    delay_ms(2);

    // Function Set Commands
    lcd_cmd(0x28); // 4-bit bus, 2 lines, 5x8 matrix
    lcd_cmd(0x0C); // Display ON, Cursor OFF, Blink OFF
    lcd_cmd(0x01); // Clear Display Screen
    delay_ms(5);
    lcd_cmd(0x06); // Entry Mode: Increment Cursor automatically
}
