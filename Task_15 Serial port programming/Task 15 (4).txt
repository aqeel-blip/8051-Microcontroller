#include <reg52.h>

#define MAX_LEN 16

// LCD Pin Definitions for A2 Development Board
sbit LCD_RS = P2^6;
sbit LCD_WR = P2^5;
sbit LCD_E  = P2^7;
#define LCD_DATA P0

// Global Variables
char rx_buffer[MAX_LEN + 1];
unsigned char rx_index = 0;
bit msg_received = 0;

// Parsed Time Variables (Stored as requested)
unsigned char hour = 0;
unsigned char min  = 0;
unsigned char sec  = 0;

// Delay Utility
void Delay_ms(unsigned int ms) {
    unsigned int i, j;
    for (i = 0; i < ms; i++)
        for (j = 0; j < 120; j++);
}

// LCD Functions
void LCD_Command(unsigned char cmd) {
    LCD_RS = 0;
    LCD_WR = 0;
    LCD_DATA = cmd;
    LCD_E = 1;
    Delay_ms(2);
    LCD_E = 0;
}

void LCD_Char(unsigned char dat) {
    LCD_RS = 1;
    LCD_WR = 0;
    LCD_DATA = dat;
    LCD_E = 1;
    Delay_ms(2);
    LCD_E = 0;
}

void LCD_Init(void) {
    LCD_Command(0x38); // 8-bit, 2 lines, 5x7 font
    LCD_Command(0x0C); // Display ON, Cursor OFF
    LCD_Command(0x06); // Auto increment cursor
    LCD_Command(0x01); // Clear display
    Delay_ms(5);
}

void LCD_String(char *str) {
    while (*str) LCD_Char(*str++);
}

void LCD_SetCursor(unsigned char row, unsigned char col) {
    if (row == 0) LCD_Command(0x80 + col);
    else if (row == 1) LCD_Command(0xC0 + col);
}

// Helper function to display numbers as 2 digits on LCD
void LCD_PrintTwoDigits(unsigned char val) {
    LCD_Char((val / 10) + '0');
    LCD_Char((val % 10) + '0');
}

// UART Functions
void UART_Init(void) {
    SCON = 0x50;  // Mode 1: 8-bit UART, receive enabled
    TMOD |= 0x20; // Timer 1, Mode 2 (8-bit auto-reload)
    TH1  = 0xFD;  // 9600 baud rate @ 11.0592 MHz
    TL1  = 0xFD;
    TR1  = 1;     // Start Timer 1
    ES   = 1;     // Enable Serial Interrupt
    EA   = 1;     // Enable Global Interrupt
}

void UART_SendChar(char ch) {
    SBUF = ch;
    while (TI == 0);
    TI = 0;
}

void UART_SendString(char *str) {
    while (*str) UART_SendChar(*str++);
}

// UART Interrupt Handler
void UART_ISR(void) interrupt 4 {
    if (RI) {
        char ch = SBUF;
        RI = 0;
        
        if (!msg_received) {
            // Ignore stray newline characters at start
            if (rx_index == 0 && (ch == '\n' || ch == '\r')) {
                return;
            }
            
            // Check for line end sequence (\n or \r)
            if (ch == '\n' || ch == '\r') {
                rx_buffer[rx_index] = '\0'; // Null-terminate string
                msg_received = 1;
            } else {
                rx_buffer[rx_index] = ch;
                rx_index++;
                if (rx_index >= MAX_LEN) {
                    rx_buffer[MAX_LEN] = '\0';
                    msg_received = 1;
                }
            }
        }
    }
}

// Function to Parse String "hh:mm:ss" into hour, min, sec variables
void Parse_Time_String(void) {
    // Basic format validation: hh:mm:ss (length must be at least 8)
    if (rx_index >= 8 && rx_buffer[2] == ':' && rx_buffer[5] == ':') {
        hour = (rx_buffer[0] - '0') * 10 + (rx_buffer[1] - '0');
        min  = (rx_buffer[3] - '0') * 10 + (rx_buffer[4] - '0');
        sec  = (rx_buffer[6] - '0') * 10 + (rx_buffer[7] - '0');
        
        // Confirmation feedback to PC via Serial Terminal
        UART_SendString("Parsed -> H:");
        UART_SendChar((hour / 10) + '0'); UART_SendChar((hour % 10) + '0');
        UART_SendString(" M:");
        UART_SendChar((min / 10) + '0');  UART_SendChar((min % 10) + '0');
        UART_SendString(" S:");
        UART_SendChar((sec / 10) + '0');  UART_SendChar((sec % 10) + '0');
        UART_SendString("\r\n");
    } else {
        UART_SendString("Invalid Format! Use hh:mm:ss\\n\\r\r\n");
    }
}

void main(void) {
    LCD_Init();
    UART_Init();
    
    LCD_SetCursor(0, 0);
    LCD_String("Time Receiver");
    LCD_SetCursor(1, 0);
    LCD_String("T: --:--:--");

    while (1) {
        if (msg_received) {
            // Parse time string into variables
            Parse_Time_String();
            
            // Display parsed values on LCD
            LCD_SetCursor(1, 3);
            LCD_PrintTwoDigits(hour);
            LCD_Char(':');
            LCD_PrintTwoDigits(min);
            LCD_Char(':');
            LCD_PrintTwoDigits(sec);
            
            // Reset state
            rx_index = 0;
            msg_received = 0;
        }
    }
}
