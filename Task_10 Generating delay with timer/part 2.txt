#include <REG52.H>

// Software counters and flag declarations
int cntr_100ms = 0;
int cntr_1000ms = 0;
char flag_100_ms = 0;
char flag_1000_ms = 0;

// LED pin definitions mapped to Port 2
sbit LED_D1 = P2^0;
sbit LED_D2 = P2^1;
sbit LED_D3 = P2^2;
sbit LED_D4 = P2^3;

// Timer 0 Initialization Function (10 ms interrupt setup)
void Timer0_Init(void)
{
    TMOD &= 0xF0;     // Clear Timer 0 mode bits
    TMOD |= 0x01;     // Set Timer 0 to Mode 1 (16-bit timer)
    
    // 10 ms reload value for 12 MHz crystal clock (65536 - 10000 = 55536 -> 0xD8F0)
    TH0 = 0xD8;       
    TL0 = 0xF0;
    
    ET0 = 1;          // Enable Timer 0 Interrupt
    EA = 1;           // Enable Global Interrupts
    TR0 = 1;          // Start Timer 0
}

// Interrupt Service Routine (Executes every 10 ms)
void ISR_timer0(void) interrupt 1
{
    // Reload Timer 0 registers for next 10 ms cycle
    TH0 = 0xD8;
    TL0 = 0xF0;

    cntr_100ms++;
    cntr_1000ms++;

    // Check if counter reaches 10 (100 ms passed)
    if (cntr_100ms >= 10)
    {
        LED_D1 = ~LED_D1;      // Toggle LED D1
        flag_100_ms = 1;       // Raise the 100 ms flag
        cntr_100ms = 0;        // Restart counter
    }

    // Check if counter reaches 100 (1000 ms passed)
    if (cntr_1000ms >= 100)
    {
        LED_D2 = ~LED_D2;      // Toggle LED D2
        cntr_1000ms = 0;       // Restart counter
        flag_1000_ms = 1;      // Raise the 1000 ms flag
    }
}

void main(void)
{
    // Initialize Port 2 pins high (LEDs initial state OFF)
    P2 = 0xFF;

    Timer0_Init();

    while (1)
    {
        // Execute block every 100 ms via flag
        if (flag_100_ms)
        {
            flag_100_ms = 0;
            LED_D3 = ~LED_D3;   // Toggle LED D3
        }

        // Execute block every 1000 ms via flag
        if (flag_1000_ms)
        {
            flag_1000_ms = 0;
            LED_D4 = ~LED_D4;   // Toggle LED D4
        }
    }
}
